% =========================================================
% 全国大学生数学建模竞赛论文排版设置
% 所有排版集中在这里；正文填写各板块 content.tex
% =========================================================

% =========================
% 页面尺寸与页边距
% =========================
\usepackage[
  a4paper,
  top=2.2cm,
  bottom=2.2cm,
  left=2.2cm,
  right=2.2cm,
  includehead=false,
  includefoot=false
]{geometry}

% =========================
% 数学公式
% =========================
\usepackage{amsmath,amssymb,bm}

% =========================
% 图片与表格
% =========================
\usepackage{graphicx,booktabs,tabularx,longtable,array}
\usepackage{caption,subcaption,float,placeins}
\usepackage[table]{xcolor}

% =========================
% 列表、页眉页脚、代码
% =========================
\usepackage{enumitem,fancyhdr,fvextra}

% =========================
% 超链接
% =========================
\usepackage[hidelinks,unicode]{hyperref}

% 图片默认路径
\graphicspath{{../Figures/}}

% =========================================================
% 正文基础排版
% =========================================================

% 首行缩进 2 个汉字
\setlength{\parindent}{2em}

% 段间距
\setlength{\parskip}{0pt}

% 行距
\linespread{1.25}

% =========================================================
% 标题格式
% =========================================================

\ctexset{
  section={
    format=\centering\heiti\zihao{4},
    name={,、},
    number=\chinese{section},
    aftername=\hspace{0.3em},
    beforeskip=16pt,
    afterskip=9pt
  },
  subsection={
    format=\heiti\zihao{-4},
    number=\arabic{section}.\arabic{subsection},
    beforeskip=12pt,
    afterskip=6pt
  },
  subsubsection={
    format=\heiti\zihao{-4},
    number=\arabic{section}.\arabic{subsection}.\arabic{subsubsection},
    beforeskip=8pt,
    afterskip=4pt
  }
}

% 标题编号深度
\setcounter{secnumdepth}{3}

% =========================================================
% 页码设置
% =========================================================

\pagestyle{fancy}

% 清空默认页眉页脚
\fancyhf{}

% 页脚正中央页码
\fancyfoot[C]{\zihao{5}\thepage}

% 去除页眉横线
\renewcommand{\headrulewidth}{0pt}

% 去除页脚横线
\renewcommand{\footrulewidth}{0pt}

% 页码与正文距离
\setlength{\footskip}{1.5cm}

% =========================================================
% 图表设置
% =========================================================

% 图表标题字号、间距
\captionsetup{
  font=small,
  labelsep=quad,
  skip=6pt
}

% 表格标题位于表格上方
\captionsetup[table]{
  position=top
}

% 表格行距
\renewcommand{\arraystretch}{1.2}

% 浮动体与正文间距
\setlength{\textfloatsep}{12pt plus 2pt minus 2pt}
\setlength{\intextsep}{10pt plus 2pt minus 2pt}

% =========================================================
% 列表设置
% =========================================================

\setlist{
  nosep,
  topsep=5pt,
  leftmargin=2em
}

\setlist[enumerate]{
  label=(\arabic*)
}

% =========================================================
% 代码环境设置
% =========================================================

\fvset{
  fontsize=\small,
  breaklines=true,
  breakanywhere=true,
  frame=single,
  numbers=left,
  numbersep=5pt,
  xleftmargin=1.5em
}

% =========================================================
% 图片占位框
% =========================================================

\newcommand{\figureplaceholder}[2]{%
  \fbox{%
    \parbox[c][35mm][c]{0.85\linewidth}{%
      \centering
      \heiti #1\par
      \vspace{8pt}
      \normalfont\small #2
    }%
  }%
}

% =========================================================
% 公式上下间距
% =========================================================

\AtBeginDocument{%
  \setlength{\abovedisplayskip}{8pt plus 2pt minus 2pt}
  \setlength{\belowdisplayskip}{8pt plus 2pt minus 2pt}
}

% 避免 LaTeX 为了填满整页而强行拉大段间距
\raggedbottom

% =========================================================
% 摘要首页标题
% 论文标题：三号黑体约 16pt
% “摘要”：四号黑体约 14pt
% =========================================================

\newcommand{\makepaperheading}[1]{%
  \par
  \begingroup
  \centering

  {\heiti\fontsize{16pt}{22pt}\selectfont
    #1\par
  }

  \vspace{12pt}

  {\heiti\fontsize{14pt}{20pt}\selectfont
    摘\quad 要\par
  }

  \endgroup

  \vspace{10pt}
}
